cmake_minimum_required(VERSION 3.25)
project(LumaWhisperWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LUMA_BUILD_WORKER "Build the ASR worker and pinned Whisper library" ON)

option(LUMA_CPU_ONLY_DEVELOPER_BUILD "Disable GPU and CPU multivariants in this build tree" OFF)
option(LUMA_BUILD_CAPTION_TESTS "Build model-free native caption contract tests" ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()

function(luma_register_native_test name target)
  add_test(NAME ${name} COMMAND ${target} ${ARGN})
  if(WIN32)
    get_filename_component(ov_runtime_root "${OpenVINO_DIR}/.." ABSOLUTE)
    set_property(TEST ${name} PROPERTY ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:$<TARGET_FILE_DIR:openvino::runtime>"
      "PATH=path_list_prepend:${ov_runtime_root}/3rdparty/tbb/bin")
  endif()
endfunction()

if(LUMA_BUILD_WORKER OR LUMA_BUILD_ENCODER_TOOLS)
file(TO_CMAKE_PATH "${WHISPER_SOURCE}" WHISPER_SOURCE)
if(NOT EXISTS "${WHISPER_SOURCE}/include/whisper.h")
  message(FATAL_ERROR "Set WHISPER_SOURCE to the pinned whisper.cpp source")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(WHISPER_OPENVINO ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
if(LUMA_CPU_ONLY_DEVELOPER_BUILD)
  set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
  set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
else()
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  set(GGML_VULKAN ON CACHE BOOL "" FORCE)
endif()
file(READ "${WHISPER_SOURCE}/src/whisper.cpp" whisper_source)
string(FIND "${whisper_source}" "// LUMA_BACKEND_INFO_BEGIN" extension_start)
if(NOT extension_start EQUAL -1)
  string(SUBSTRING "${whisper_source}" 0 ${extension_start} whisper_source)
  string(REGEX REPLACE "[\r\n]+$" "" whisper_source "${whisper_source}")
endif()
# The b5454 external-encoder branch ignores the adapter's failure result.
# Apply exactly the known change, or fail configuration if upstream drifted.
set(openvino_unchecked "                whisper_openvino_encode(wstate.ctx_openvino, mel, wstate.embd_enc);")
set(openvino_checked "                if (!whisper_openvino_encode(wstate.ctx_openvino, mel, wstate.embd_enc)) {\n                    return false;\n                }")
string(FIND "${whisper_source}" "${openvino_unchecked}" openvino_unchecked_start)
if(NOT openvino_unchecked_start EQUAL -1)
  string(REPLACE "${openvino_unchecked}" "${openvino_checked}" whisper_source "${whisper_source}")
else()
  string(FIND "${whisper_source}" "${openvino_checked}" openvino_checked_start)
  if(openvino_checked_start EQUAL -1)
    message(FATAL_ERROR "Pinned Whisper OpenVINO failure branch changed; review the patch before building")
  endif()
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/backend_info.inc" extension)
set(patched_source "${whisper_source}\n${extension}")
file(READ "${WHISPER_SOURCE}/src/whisper.cpp" previous_source)
if(NOT patched_source STREQUAL previous_source)
  file(WRITE "${WHISPER_SOURCE}/src/whisper.cpp" "${patched_source}")
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper)
