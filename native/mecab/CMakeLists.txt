cmake_minimum_required(VERSION 3.25)
project(LumaMeCab LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This adapter is for MSVC; use the system MeCab on Linux")
endif()
set(CMAKE_CXX_STANDARD 11)
file(TO_CMAKE_PATH "${MECAB_SOURCE}" MECAB_SOURCE)
if(NOT EXISTS "${MECAB_SOURCE}/src/libmecab.cpp")
  message(FATAL_ERROR "Set MECAB_SOURCE to the pinned upstream mecab directory")
endif()
# Keep the upstream tree intact. The 2015 MSVC header omits WPATH_FORCE
# outside the GCC branch; restore its UTF-8→UTF-16 path conversion locally.
file(GLOB headers "${MECAB_SOURCE}/src/*.h")
foreach(header IN LISTS headers)
  get_filename_component(name "${header}" NAME)
  configure_file("${header}" "${CMAKE_CURRENT_BINARY_DIR}/source/${name}" COPYONLY)
endforeach()
file(READ "${CMAKE_CURRENT_BINARY_DIR}/source/common.h" common)
string(REPLACE "#ifdef __GNUC__\n#define WPATH_FORCE(path) (MeCab::Utf8ToWide(path).c_str())" "#define WPATH_FORCE(path) (MeCab::Utf8ToWide(path).c_str())\n#ifdef __GNUC__" common "${common}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source/common.h" "#include <functional>\n${common}")
set(units feature_index param learner string_buffer char_property learner_tagger tagger connector tokenizer context_id dictionary utils dictionary_compiler viterbi dictionary_generator writer iconv_utils dictionary_rewriter lbfgs eval nbest_generator libmecab)
foreach(unit IN LISTS units)
  configure_file("${MECAB_SOURCE}/src/${unit}.cpp" "${CMAKE_CURRENT_BINARY_DIR}/source/${unit}.cpp" COPYONLY)
  list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/source/${unit}.cpp")
endforeach()
add_library(mecab SHARED ${sources})
target_compile_definitions(mecab PRIVATE _CRT_SECURE_NO_WARNINGS _HAS_AUTO_PTR_ETC=1 MECAB_USE_THREAD DLL_EXPORT HAVE_GETENV HAVE_WINDOWS_H HAVE_UNSIGNED_LONG_LONG_INT DIC_VERSION=102 VERSION="0.996" PACKAGE="mecab" UNICODE _UNICODE MECAB_DEFAULT_RC="mecabrc")
target_compile_options(mecab PRIVATE /EHsc /wd4244 /wd4267 /wd4305 /wd4800)
target_link_libraries(mecab PRIVATE advapi32)
add_executable(mecab-dict-index "${MECAB_SOURCE}/src/mecab-dict-index.cpp")
target_include_directories(mecab-dict-index PRIVATE "${MECAB_SOURCE}/src")
target_link_libraries(mecab-dict-index PRIVATE mecab)
