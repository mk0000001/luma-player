cmake_minimum_required(VERSION 3.25)
project(LumaTranslationTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
endif()
get_filename_component(LUMA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(luma_translation "${LUMA_ROOT}/src/subtitles/translation_support.cpp" "${LUMA_ROOT}/src/subtitles/translation_service.cpp" "${LUMA_ROOT}/src/subtitles/translation_service.h")
target_include_directories(luma_translation PUBLIC "${LUMA_ROOT}/src")
target_link_libraries(luma_translation PUBLIC Qt6::Core Qt6::Network)
add_executable(translation_tests "${LUMA_ROOT}/tests/translation_tests.cpp")
target_link_libraries(translation_tests PRIVATE luma_translation Qt6::Test)
enable_testing()
add_test(NAME translation COMMAND translation_tests)
add_executable(translation_worker_fixture "${LUMA_ROOT}/tests/translation_worker_fixture.cpp")
target_link_libraries(translation_worker_fixture PRIVATE Qt6::Core Qt6::Network)
add_executable(translation_service_tests "${LUMA_ROOT}/tests/translation_service_tests.cpp")
target_link_libraries(translation_service_tests PRIVATE luma_translation Qt6::Test)
add_dependencies(translation_service_tests translation_worker_fixture)
add_test(NAME translation_service COMMAND translation_service_tests)
add_executable(translation_native_tests "${LUMA_ROOT}/tests/translation_native_tests.cpp")
target_link_libraries(translation_native_tests PRIVATE luma_translation Qt6::Test)
add_test(NAME translation_native_opt_in COMMAND translation_native_tests)
