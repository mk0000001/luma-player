cmake_minimum_required(VERSION 3.25)
project(LumaPlayer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(LUMA_BUILD_APP "Build the native player" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(luma_core src/core/disc.cpp src/core/subtitle.cpp)
target_include_directories(luma_core PUBLIC src)
target_link_libraries(luma_core PUBLIC Qt6::Core)
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
target_link_libraries(core_tests PRIVATE luma_core Qt6::Test)
add_test(NAME core COMMAND core_tests)
if(LUMA_BUILD_APP)
  find_path(VLC_INCLUDE_DIR vlc/vlc.h HINTS "${CMAKE_SOURCE_DIR}/.deps/vlc/vlc-3.0.24/sdk/include")
  find_path(MPV_INCLUDE_DIR mpv/client.h HINTS "${CMAKE_SOURCE_DIR}/.deps/mpv/include")
  if(NOT VLC_INCLUDE_DIR OR NOT MPV_INCLUDE_DIR)
    message(FATAL_ERROR "Media headers missing. Run bootstrap-windows.ps1 or install libvlc-dev and libmpv-dev.")
  endif()
  add_library(luma_media src/media/playback.cpp src/media/playback.h)
  target_include_directories(luma_media PRIVATE "${VLC_INCLUDE_DIR}" "${MPV_INCLUDE_DIR}")
  target_link_libraries(luma_media PUBLIC luma_core Qt6::Widgets)
  add_executable(media_tests tests/media_tests.cpp)
  target_link_libraries(media_tests PRIVATE luma_media Qt6::Test)
  add_test(NAME media COMMAND media_tests)
  set_tests_properties(media PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
