cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LUMA_VERSION)
string(STRIP "${LUMA_VERSION}" LUMA_VERSION)
if(NOT LUMA_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.]+)?$")
  message(FATAL_ERROR "VERSION must contain a semantic version")
endif()
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" LUMA_BASE_VERSION "${LUMA_VERSION}")
project(LumaPlayer VERSION ${LUMA_BASE_VERSION} LANGUAGES CXX)
set(LUMA_BUILD_NUMBER 0)
if(LUMA_VERSION MATCHES "-.*\\.([0-9]+)$")
  set(LUMA_BUILD_NUMBER "${CMAKE_MATCH_1}")
endif()
configure_file(src/version.h.in generated/luma_version.h @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(LUMA_BUILD_APP "Build the native player" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(luma_core src/core/disc.cpp src/core/subtitle.cpp)
target_include_directories(luma_core PUBLIC src)
target_include_directories(luma_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(luma_core PUBLIC Qt6::Core)
add_library(luma_subtitles src/subtitles/subtitle_job.cpp src/subtitles/subtitle_job.h src/subtitles/pcm_buffer.cpp src/subtitles/pcm_buffer.h src/subtitles/japanese_reading.cpp src/subtitles/ai_backend.cpp src/subtitles/ai_devices.cpp src/subtitles/ai_devices.h)
target_sources(luma_subtitles PRIVATE src/subtitles/translation_service.cpp src/subtitles/translation_service.h src/subtitles/translation_support.cpp src/subtitles/translation_support.h)
target_link_libraries(luma_subtitles PUBLIC luma_core Qt6::Network)
add_executable(translation_tests tests/translation_tests.cpp)
target_link_libraries(translation_tests PRIVATE luma_subtitles Qt6::Test)
add_test(NAME translation COMMAND translation_tests)
add_executable(translation_worker_fixture tests/translation_worker_fixture.cpp)
target_link_libraries(translation_worker_fixture PRIVATE Qt6::Core Qt6::Network)
add_executable(translation_service_tests tests/translation_service_tests.cpp)
target_link_libraries(translation_service_tests PRIVATE luma_subtitles Qt6::Test)
add_dependencies(translation_service_tests translation_worker_fixture)
add_test(NAME translation_service COMMAND translation_service_tests)
add_executable(translation_native_tests tests/translation_native_tests.cpp)
target_link_libraries(translation_native_tests PRIVATE luma_subtitles Qt6::Test)
add_test(NAME translation_native_opt_in COMMAND translation_native_tests)
add_executable(reading_tests tests/reading_tests.cpp)
target_link_libraries(reading_tests PRIVATE luma_subtitles Qt6::Test)
add_test(NAME readings COMMAND reading_tests)
add_executable(ai_backend_tests tests/ai_backend_tests.cpp)
target_link_libraries(ai_backend_tests PRIVATE luma_subtitles Qt6::Test)
add_test(NAME ai_backends COMMAND ai_backend_tests)
add_executable(subtitle_job_tests tests/subtitle_job_tests.cpp)
add_executable(subtitle_worker_fixture tests/subtitle_worker_fixture.cpp)
set_target_properties(subtitle_worker_fixture PROPERTIES AUTOMOC OFF)
add_dependencies(subtitle_job_tests subtitle_worker_fixture)
add_executable(pcm_buffer_tests tests/pcm_buffer_tests.cpp)
target_link_libraries(pcm_buffer_tests PRIVATE luma_subtitles Qt6::Test)
add_dependencies(pcm_buffer_tests subtitle_worker_fixture)
add_test(NAME pcm_buffer COMMAND pcm_buffer_tests)
target_link_libraries(subtitle_job_tests PRIVATE luma_subtitles Qt6::Test)
add_test(NAME subtitle_jobs COMMAND subtitle_job_tests -o "${CMAKE_CURRENT_BINARY_DIR}/subtitle-test-results.txt,txt")
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
target_link_libraries(core_tests PRIVATE luma_core Qt6::Test)
add_test(NAME core COMMAND core_tests)
if(LUMA_BUILD_APP)
  find_path(VLC_INCLUDE_DIR vlc/vlc.h HINTS "${CMAKE_SOURCE_DIR}/.deps/vlc/vlc-3.0.24/sdk/include")
  find_path(MPV_INCLUDE_DIR mpv/client.h HINTS "${CMAKE_SOURCE_DIR}/.deps/mpv/include")
  if(NOT VLC_INCLUDE_DIR OR NOT MPV_INCLUDE_DIR)
    message(FATAL_ERROR "Media headers missing. Run bootstrap-windows.ps1 or install libvlc-dev and libmpv-dev.")
  endif()
  add_library(luma_media src/media/playback.cpp src/media/playback.h src/media/platform_features.cpp src/media/platform_features.h)
  target_sources(luma_media PRIVATE src/media/motion_options.cpp src/media/motion_options.h)
  add_executable(motion_tests tests/motion_tests.cpp)
  target_link_libraries(motion_tests PRIVATE luma_media Qt6::Test)
  add_test(NAME motion COMMAND motion_tests)
  target_include_directories(luma_media PRIVATE "${VLC_INCLUDE_DIR}" "${MPV_INCLUDE_DIR}")
  target_link_libraries(luma_media PUBLIC luma_core Qt6::Widgets)
  if(WIN32)
    target_link_libraries(luma_media PRIVATE dxgi dwmapi)
  endif()
  add_executable(platform_tests tests/platform_tests.cpp)
  target_link_libraries(platform_tests PRIVATE luma_media Qt6::Test)
  add_test(NAME platform COMMAND platform_tests)
  set_tests_properties(platform PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    set_tests_properties(platform PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_SOURCE_DIR}/.deps/qt/plugins")
  endif()
  add_executable(media_tests tests/media_tests.cpp)
  target_include_directories(media_tests PRIVATE "${VLC_INCLUDE_DIR}")
  target_link_libraries(media_tests PRIVATE luma_media Qt6::Test)
  add_test(NAME media COMMAND media_tests)
  set_tests_properties(media PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set(CMAKE_AUTORCC ON)
  add_library(luma_ui src/ui/main_window.cpp src/ui/main_window.h src/ui/caption_overlay.cpp src/ui/caption_overlay.h src/ui/theme.cpp src/ui/player_controls.cpp src/ui/player_controls.h src/ui/fullscreen_controls.cpp src/ui/fullscreen_controls.h assets/luma.qrc)
  target_link_libraries(luma_ui PUBLIC luma_media luma_subtitles Qt6::Widgets)
  qt_add_executable(LumaPlayer WIN32 src/main.cpp)
  if(WIN32)
    enable_language(RC)
    configure_file(src/windows_version.rc.in generated/luma-version.rc @ONLY)
    target_sources(LumaPlayer PRIVATE assets/luma.rc "${CMAKE_CURRENT_BINARY_DIR}/generated/luma-version.rc")
    target_include_directories(LumaPlayer PRIVATE assets)
  endif()
  target_link_libraries(LumaPlayer PRIVATE luma_ui)
  add_executable(ui_tests tests/ui_tests.cpp)
  target_link_libraries(ui_tests PRIVATE luma_ui Qt6::Test)
  add_test(NAME ui_keyboard COMMAND ui_tests -o "${CMAKE_CURRENT_BINARY_DIR}/ui-test-results.txt,txt")
  add_executable(native_playback_tests tests/native_playback_tests.cpp)
  target_link_libraries(native_playback_tests PRIVATE luma_media luma_subtitles Qt6::Test)
  add_test(NAME native_opt_in COMMAND native_playback_tests)
  set_tests_properties(native_opt_in PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(ui_keyboard PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    set_tests_properties(ui_keyboard native_opt_in PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_SOURCE_DIR}/.deps/qt/plugins")
  endif()
  if(WIN32)
    target_link_libraries(luma_ui PRIVATE dwmapi)
  endif()
  install(TARGETS LumaPlayer RUNTIME DESTINATION .)
endif()
