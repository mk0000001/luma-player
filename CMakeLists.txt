cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LUMA_VERSION)
string(STRIP "${LUMA_VERSION}" LUMA_VERSION)
if(NOT LUMA_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.]+)?$")
  message(FATAL_ERROR "VERSION must contain a semantic version")
endif()
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" LUMA_BASE_VERSION "${LUMA_VERSION}")
project(LumaPlayer VERSION ${LUMA_BASE_VERSION} LANGUAGES CXX)
configure_file(src/version.h.in generated/luma_version.h @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(LUMA_BUILD_APP "Build the native player" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(luma_core src/core/disc.cpp src/core/subtitle.cpp)
target_include_directories(luma_core PUBLIC src)
target_include_directories(luma_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(luma_core PUBLIC Qt6::Core)
add_library(luma_subtitles src/subtitles/subtitle_job.cpp src/subtitles/subtitle_job.h)
target_link_libraries(luma_subtitles PUBLIC luma_core)
add_executable(subtitle_job_tests tests/subtitle_job_tests.cpp)
add_executable(subtitle_worker_fixture tests/subtitle_worker_fixture.cpp)
set_target_properties(subtitle_worker_fixture PROPERTIES AUTOMOC OFF)
add_dependencies(subtitle_job_tests subtitle_worker_fixture)
target_link_libraries(subtitle_job_tests PRIVATE luma_subtitles Qt6::Test)
add_test(NAME subtitle_jobs COMMAND subtitle_job_tests)
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
target_link_libraries(core_tests PRIVATE luma_core Qt6::Test)
add_test(NAME core COMMAND core_tests)
if(LUMA_BUILD_APP)
  find_path(VLC_INCLUDE_DIR vlc/vlc.h HINTS "${CMAKE_SOURCE_DIR}/.deps/vlc/vlc-3.0.24/sdk/include")
  find_path(MPV_INCLUDE_DIR mpv/client.h HINTS "${CMAKE_SOURCE_DIR}/.deps/mpv/include")
  if(NOT VLC_INCLUDE_DIR OR NOT MPV_INCLUDE_DIR)
    message(FATAL_ERROR "Media headers missing. Run bootstrap-windows.ps1 or install libvlc-dev and libmpv-dev.")
  endif()
  add_library(luma_media src/media/playback.cpp src/media/playback.h)
  target_include_directories(luma_media PRIVATE "${VLC_INCLUDE_DIR}" "${MPV_INCLUDE_DIR}")
  target_link_libraries(luma_media PUBLIC luma_core Qt6::Widgets)
  add_executable(media_tests tests/media_tests.cpp)
  target_include_directories(media_tests PRIVATE "${VLC_INCLUDE_DIR}")
  target_link_libraries(media_tests PRIVATE luma_media Qt6::Test)
  add_test(NAME media COMMAND media_tests)
  set_tests_properties(media PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_library(luma_ui src/ui/main_window.cpp src/ui/main_window.h)
  target_link_libraries(luma_ui PUBLIC luma_media luma_subtitles Qt6::Widgets)
  qt_add_executable(LumaPlayer WIN32 src/main.cpp)
  target_link_libraries(LumaPlayer PRIVATE luma_ui)
  add_executable(ui_tests tests/ui_tests.cpp)
  target_link_libraries(ui_tests PRIVATE luma_ui Qt6::Test)
  add_test(NAME ui_keyboard COMMAND ui_tests)
  add_executable(native_playback_tests tests/native_playback_tests.cpp)
  target_link_libraries(native_playback_tests PRIVATE luma_media luma_subtitles Qt6::Test)
  add_test(NAME native_opt_in COMMAND native_playback_tests)
  set_tests_properties(native_opt_in PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(ui_keyboard PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    set_tests_properties(ui_keyboard native_opt_in PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_SOURCE_DIR}/.deps/qt/plugins")
  endif()
  if(WIN32)
    target_link_libraries(luma_ui PRIVATE dwmapi)
  endif()
  install(TARGETS LumaPlayer RUNTIME DESTINATION .)
endif()
